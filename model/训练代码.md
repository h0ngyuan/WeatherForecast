## 温度模型训练代码
```jupyter
import pandas as pd
import numpy as np
import torch
import lightning.pytorch as pl
from lightning.pytorch import Trainer
from lightning.pytorch.loggers import CSVLogger
from lightning.pytorch.callbacks import EarlyStopping, LearningRateMonitor
from pytorch_forecasting import (
    TimeSeriesDataSet,
    TemporalFusionTransformer,
    QuantileLoss,
    GroupNormalizer
)
from sklearn.preprocessing import StandardScaler
import warnings
import pickle
import os

warnings.filterwarnings("ignore")
pl.seed_everything(42)
csv_logger = CSVLogger(save_dir="/kaggle/working/logs", name="tft_weather")

# ==============================
# 1. 列名映射
# ==============================
COLUMN_MAPPING = {
    "temp": "temperature_2m (°C)",
    "rh": "relative_humidity_2m (%)",
    "dew": "dew_point_2m (°C)",
    "precip": "precipitation (mm)",
    "rain": "rain (mm)",
    "snow": "snowfall (cm)",
    "app_temp": "apparent_temperature (°C)",
    "weather_code": "weather_code (wmo code)",
    "p_msl": "pressure_msl (hPa)",
    "surf_p": "surface_pressure (hPa)",
    "cloud": "cloud_cover (%)",
    "wind10": "wind_speed_10m (km/h)",
    "dir10": "wind_direction_10m (°)",
    "gust10": "wind_gusts_10m (km/h)",
    "soil_t0": "soil_temperature_0_to_7cm (°C)",
    "snow_depth": "snow_depth (m)",
    "cloud_low": "cloud_cover_low (%)",
    "cloud_mid": "cloud_cover_mid (%)",
    "cloud_high": "cloud_cover_high (%)",
    "et0": "et0_fao_evapotranspiration (mm)",
    "vpd": "vapour_pressure_deficit (kPa)",
    "wind100": "wind_speed_100m (km/h)",
    "dir100": "wind_direction_100m (°)",
    "soil_t7": "soil_temperature_7_to_28cm (°C)",
    "soil_m0": "soil_moisture_0_to_7cm (m³/m³)",
    "soil_m7": "soil_moisture_7_to_28cm (m³/m³)"
}
REVERSE_MAPPING = {v: k for k, v in COLUMN_MAPPING.items()}

# ==============================
# 2. 加载数据
# ==============================
clean_path = "你要用的csv文件"
df = pd.read_csv(clean_path)
df['time'] = pd.to_datetime(df['time'])
df = df.sort_values("time").reset_index(drop=True)
df['time_idx'] = np.arange(len(df))
df = df.rename(columns=REVERSE_MAPPING)
NUMERICAL_COLUMNS = list(COLUMN_MAPPING.keys())

# ==============================
# 3. 特征定义
# ==============================
target = "temp"
known_features = ["wind10", "dir10", "p_msl", "cloud", "wind100", "dir100"]
unknown_features = [col for col in NUMERICAL_COLUMNS if col != target and col not in known_features]

# ==============================
# 4. 划分训练/验证集
# ==============================
max_encoder_length = 168      # 输入过去 7 天 (168 小时)
max_prediction_length = 24     # 预测未来 24 小时
val_size = 720  # 验证集大小
min_required = max_encoder_length + max_prediction_length
cutoff = df["time_idx"].max() - val_size
train_df = df[df["time_idx"] <= cutoff].copy()
val_df = df[df["time_idx"] > cutoff].copy()
train_df["group"] = 0
val_df["group"] = 0

# ==============================
# 5. 标准化（仅对非恒定列）
# ==============================
cols_to_scale = NUMERICAL_COLUMNS.copy()
stds_train = train_df[cols_to_scale].std()
valid_cols = [col for col in cols_to_scale if stds_train[col] > 1e-8]

scaler = None
if valid_cols:
    scaler = StandardScaler()
    train_df[valid_cols] = scaler.fit_transform(train_df[valid_cols])
    val_df[valid_cols] = scaler.transform(val_df[valid_cols])

with open("/kaggle/working/scaler.pkl", "wb") as f:
    pickle.dump((scaler, valid_cols), f)

# ==============================
# 6. 构建 TimeSeriesDataSet
# ==============================
dataset_train = TimeSeriesDataSet(
    train_df,
    time_idx="time_idx",
    target=target,
    group_ids=["group"],
    min_encoder_length=max_encoder_length,
    max_encoder_length=max_encoder_length,
    min_prediction_length=max_prediction_length,
    max_prediction_length=max_prediction_length,
    static_categoricals=[],
    static_reals=[],
    time_varying_known_reals=known_features,
    time_varying_unknown_reals=[target] + unknown_features,
    target_normalizer=GroupNormalizer(groups=["group"]),
    add_relative_time_idx=True,
    add_target_scales=True,
    add_encoder_length=True,
    allow_missing_timesteps=False,
    lags={},
)
dataset_val = TimeSeriesDataSet.from_dataset(dataset_train, val_df, predict=True)

# ==============================
# 7. 模型配置
# ==============================
BATCH_SIZE = 256
model = TemporalFusionTransformer.from_dataset(
    dataset_train,
    learning_rate=0.03,
    hidden_size=64,  
    attention_head_size=4,
    dropout=0.1,
    hidden_continuous_size=64,
    output_size=7,   
    loss=QuantileLoss(quantiles=[0.02, 0.1, 0.25, 0.5, 0.75, 0.9, 0.98]),
    reduce_on_plateau_patience=4,
)

# ==============================
# 8. 数据加载器 & 训练器
# ==============================
train_loader = dataset_train.to_dataloader(train=True, batch_size=BATCH_SIZE, num_workers=2)
val_loader = dataset_val.to_dataloader(train=False, batch_size=BATCH_SIZE * 2, num_workers=2)

trainer = Trainer(
    accelerator="gpu",
    devices=1,
    max_epochs=50,
    gradient_clip_val=0.1,
    callbacks=[
        EarlyStopping(monitor="val_loss", patience=7, mode="min"),
        LearningRateMonitor(logging_interval="epoch"),
    ],
    enable_progress_bar=True,
    logger=csv_logger,
    default_root_dir="/kaggle/working/logs",
    precision= "32", 
)

# ==============================
# 9. 训练
# ==============================
trainer.fit(model, train_dataloaders=train_loader, val_dataloaders=val_loader)

# ==============================
# 10. 保存模型
# ==============================
trainer.save_checkpoint("/kaggle/working/自定义名称.ckpt")

x, _ = next(iter(val_loader))
model.eval()
with torch.no_grad():
    traced_model = torch.jit.trace(model, x)
traced_model.save("/kaggle/working/自定义名称.pt")
```

import pandas as pd
import numpy as np
import torch
import lightning.pytorch as pl
from lightning.pytorch import Trainer
from lightning.pytorch.loggers import CSVLogger
from lightning.pytorch.callbacks import EarlyStopping, LearningRateMonitor
from pytorch_forecasting import TimeSeriesDataSet, TemporalFusionTransformer
from pytorch_forecasting.metrics import CrossEntropy
from sklearn.preprocessing import StandardScaler
import warnings
import pickle
import os

warnings.filterwarnings("ignore")
pl.seed_everything(42)
csv_logger = CSVLogger(save_dir="/kaggle/working/logs", name="tft_weather_code")

# ==============================
# 1. 列名映射
# ==============================
COLUMN_MAPPING = {
    "temp": "temperature_2m (°C)",
    "rh": "relative_humidity_2m (%)",
    "dew": "dew_point_2m (°C)",
    "precip": "precipitation (mm)",
    "rain": "rain (mm)",
    "snow": "snowfall (cm)",
    "app_temp": "apparent_temperature (°C)",
    "weather_code": "weather_code (wmo code)",
    "p_msl": "pressure_msl (hPa)",
    "surf_p": "surface_pressure (hPa)",
    "cloud": "cloud_cover (%)",
    "wind10": "wind_speed_10m (km/h)",
    "dir10": "wind_direction_10m (°)",
    "gust10": "wind_gusts_10m (km/h)",
    "soil_t0": "soil_temperature_0_to_7cm (°C)",
    "snow_depth": "snow_depth (m)",
    "cloud_low": "cloud_cover_low (%)",
    "cloud_mid": "cloud_cover_mid (%)",
    "cloud_high": "cloud_cover_high (%)",
    "et0": "et0_fao_evapotranspiration (mm)",
    "vpd": "vapour_pressure_deficit (kPa)",
    "wind100": "wind_speed_100m (km/h)",
    "dir100": "wind_direction_100m (°)",
    "soil_t7": "soil_temperature_7_to_28cm (°C)",
    "soil_m0": "soil_moisture_0_to_7cm (m³/m³)",
    "soil_m7": "soil_moisture_7_to_28cm (m³/m³)"
}

REVERSE_MAPPING = {v: k for k, v in COLUMN_MAPPING.items()}

# ==============================
# 2. 加载数据
# ==============================
clean_path = "你要用的csv文件"

df = pd.read_csv(clean_path)
df['time'] = pd.to_datetime(df['time'])
df = df.sort_values("time").reset_index(drop=True)
df['time_idx'] = np.arange(len(df))
df = df.rename(columns=REVERSE_MAPPING)
ALL_COLUMNS = list(COLUMN_MAPPING.keys())

# ==============================
# 3. 处理 weather_code
# ==============================
original_target = "weather_code"
df[original_target] = pd.to_numeric(df[original_target], errors="coerce")

non_null = df[original_target].dropna()
mode_val = non_null.mode()
mode_val = mode_val.iloc[0] if len(mode_val) > 0 else non_null.iloc[0]
df[original_target] = df[original_target].fillna(mode_val).astype(int)

# ==============================
# 4. 特征定义 + 目标列
# ==============================
target_col = "weather_code_idx"

known_features = ["wind10", "dir10", "p_msl", "cloud", "wind100", "dir100"]
NUMERICAL_COLUMNS = [col for col in ALL_COLUMNS if col != original_target]
unknown_features = [col for col in NUMERICAL_COLUMNS if col not in known_features]
unknown_features.append(target_col)

# ==============================
# 5. 划分训练/验证集
# ==============================
max_encoder_length = 168
max_prediction_length = 24
val_size = 720
cutoff = df["time_idx"].max() - val_size
train_df = df[df["time_idx"] <= cutoff].copy()
val_df = df[df["time_idx"] > cutoff].copy()
train_df["group"] = 0
val_df["group"] = 0

wmo_codes = sorted(train_df[original_target].unique())
wmo_to_idx = {code: idx for idx, code in enumerate(wmo_codes)}
idx_to_wmo = {idx: code for code, idx in wmo_to_idx.items()}
train_df[target_col] = train_df[original_target].map(wmo_to_idx).astype(np.int64)  # ← int64!
val_df[target_col] = val_df[original_target].map(wmo_to_idx).astype(np.int64)      # ← int64!

num_classes = len(wmo_codes)

# ==============================
# 6. 标准化
# ==============================
cols_to_scale = NUMERICAL_COLUMNS.copy()
stds_train = train_df[cols_to_scale].std()
valid_cols = [col for col in cols_to_scale if stds_train[col] > 1e-8]

scaler = None
if valid_cols:
    scaler = StandardScaler()
    train_df[valid_cols] = scaler.fit_transform(train_df[valid_cols])
    val_df[valid_cols] = scaler.transform(val_df[valid_cols])

with open("/kaggle/working/scaler_weather_code.pkl", "wb") as f:
    pickle.dump((scaler, valid_cols), f)
with open("/kaggle/working/wmo_mapping.pkl", "wb") as f:
    pickle.dump((wmo_to_idx, idx_to_wmo), f)

# ==============================
# 7. 构建数据集
# ==============================
dataset_train = TimeSeriesDataSet(
    train_df,
    time_idx="time_idx",
    target=target_col,
    group_ids=["group"],
    min_encoder_length=max_encoder_length,
    max_encoder_length=max_encoder_length,
    min_prediction_length=max_prediction_length,
    max_prediction_length=max_prediction_length,
    static_categoricals=[],
    static_reals=[],
    time_varying_known_reals=known_features,
    time_varying_unknown_reals=unknown_features,
    target_normalizer=None,
    add_relative_time_idx=True,
    add_encoder_length=True,
    allow_missing_timesteps=False,
)

dataset_val = TimeSeriesDataSet.from_dataset(dataset_train, val_df, predict=False)

# ==============================
# 8. 模型配置（GPU）
# ==============================
BATCH_SIZE = 256

model = TemporalFusionTransformer.from_dataset(
    dataset_train,
    learning_rate=0.03,
    hidden_size=64,
    attention_head_size=4,
    dropout=0.1,
    hidden_continuous_size=64,
    output_size=num_classes,
    loss=CrossEntropy(),
    reduce_on_plateau_patience=4,
)

# ==============================
# 9. 数据加载器 & 训练器
# ==============================
train_loader = dataset_train.to_dataloader(train=True, batch_size=BATCH_SIZE, num_workers=2)
val_loader = dataset_val.to_dataloader(train=False, batch_size=BATCH_SIZE * 2, num_workers=2)

trainer = Trainer(
    accelerator="gpu",
    devices=1,
    max_epochs=50,
    gradient_clip_val=0.1,
    callbacks=[
        EarlyStopping(monitor="val_loss", patience=7, mode="min"),
        LearningRateMonitor(logging_interval="epoch"),
    ],
    logger=csv_logger,
    default_root_dir="/kaggle/working/logs",
    precision="32",
)

# ==============================
# 10. 训练
# ==============================
trainer.fit(model, train_dataloaders=train_loader, val_dataloaders=val_loader)

# ==============================
# 11. 保存模型（关键：现在能成功导出！）
# ==============================
trainer.save_checkpoint("/kaggle/working/自定义名称.ckpt")
